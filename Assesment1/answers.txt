from pydantic import BaseModel, Field
from typing import Annotated

class Weather(BaseModel) :
    city : Annotated[str,Field(min_length=1)]
    temperature_c: Annotated[float,Field(ge=-50, le=60)]
    humidity: Annotated[int,Field(ge=0, le=100)]
    conditions: list[str]

eg :  valid instance
weather = Weather(
    city="Jaipur",
    temperature_c=31.4,
    humidity=42,
    conditions=["sunny", "windy"]
)

weather = Weather(
    city="Jaipur",
    temperature_c=81.4,

        setError(err.message);

        setLoading(false);
      }

  const filteredUsers = users.filter((user) =>

  }

    <div>
      <h1>User Search</h1>

      <input
